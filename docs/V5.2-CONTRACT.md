# MCP Fleet v5.2 — 模块契约（并行开发用）

> 本文件由 Lead 维护。三个并行工作流各自只写「归属文件」，不得修改他人文件；
> 公共行为以本文件为准。目标是**可独立测试、可被集成方直接 import** 的纯标准库模块。

## 0. 总体约束

- Python **3.6+** 语法兼容（CI 跑 3.9 / 3.12；不得用 walrus、dataclasses、`str.removeprefix`、`asyncio.run` 等 3.7+ 特性）。
- **零第三方依赖**，只用标准库。不得 import `mcp_agent` / `fleet_proto` / `fleet`（避免循环依赖）；集成方从外部注入依赖。
- 所有落盘敏感文件权限 **0600**，目录 0700；写文件用原子写（tmp + `os.replace` + fsync）。
- 代码注释与用户可见文案用中文；标识符/协议字段用英文。所有公开函数写 docstring。
- 测试用 `pytest`，命令：`.venv/bin/python -m pytest tests/<file> -q`（venv 已建好）。
- 任何模块在依赖缺失时**降级而不是崩溃**（集成方会 try/except import）。
- 不要改 `SKILL.md`、`README.md`、`CHANGELOG.md`、`scripts/mcp_agent.py`、`scripts/fleet_proto.py`、`scripts/fleet.py`、`tests/test_*.py`（既有）。

## 1. 安全核心 `scripts/fleet_core.py`（归属：core-security）

单文件、纯标准库。对外 API（名称与签名固定）：

```python
VERSION = "5.2.4"

# ---- 身份（Ed25519）----
class Identity:
    node_id: str                 # base32(sha256(pubkey))[:16]，小写
    fingerprint: str             # "ABCD-EFGH-IJKL-MNOP" 人类可读
    public_key_b64: str
    def sign(self, msg: bytes) -> str              # base64 签名
    @classmethod
    def load_or_create(cls, path: str) -> "Identity"   # 已存在则加载，0600 落盘
    @classmethod
    def from_private_b64(cls, s: str) -> "Identity"

def verify_signature(public_key_b64: str, msg: bytes, sig_b64: str) -> bool

# ---- X25519（RFC 7748）----
def x25519(private_key: bytes, public_key: bytes) -> bytes
def x25519_keypair() -> tuple   # (priv32, pub32)

# ---- 对称加密（SHA-256 CTR 流 + HMAC-SHA256，encrypt-then-MAC）----
def hkdf_sha256(ikm: bytes, salt: bytes, info: bytes, length: int = 32) -> bytes
def seal(key: bytes, plaintext: bytes, aad: bytes = b"") -> str    # "enc2$<b64 nonce+ct+tag>"
def open_sealed(key: bytes, token: str, aad: bytes = b"") -> bytes # 失败抛 ValueError
def derive_pair_key(shared_secret: bytes, transcript: bytes) -> bytes

# ---- 请求签名 / 防重放 ----
class ReplayGuard:
    def __init__(self, window_s: int = 300, max_entries: int = 20000)
    def check(self, scope: str, nonce: str, ts: float) -> bool   # 首次 True；重放/过期 False

def sign_request(identity: Identity, method: str, path: str, body: bytes,
                 ts: float = None, nonce: str = None) -> dict
    # 返回 {"X-Fleet-Node","X-Fleet-Ts","X-Fleet-Nonce","X-Fleet-Sig"}，
    # 签名基串 = "fleet1|METHOD|PATH|ts|nonce|sha256(body) hex"

def verify_request(public_key_b64: str, method: str, path: str, body: bytes,
                   headers: dict, guard: ReplayGuard = None,
                   max_skew_s: int = 300) -> tuple   # (ok: bool, reason: str)

# ---- 邀请 / 配对（带 SAS 短认证串）----
class Invite:
    # 字段：v, name, node_id, pubkey, host, port, ts, ttl, nonce, caps, sig
    def to_uri(self) -> str       # "fleet://join/<b64url(json)>"
    @classmethod
    def parse(cls, uri: str) -> "Invite"        # 校验过期/签名，失败抛 ValueError
    @classmethod
    def create(cls, identity, name, host, port, ttl=3600, caps=None) -> "Invite"

def pair_proof(identity: Identity, invite: Invite, my_eph_priv: bytes,
               my_eph_pub: bytes, peer_eph_pub: bytes) -> tuple
    # (proof_sig_b64, sas: str, key: bytes)
    # SAS = 6 位数字，取自 sha256(transcript) 前 4 字节 % 1_000_000

# ---- 执行策略 ----
class ExecPolicy:
    def __init__(self, cfg: dict = None)      # 支持 deny_patterns/allow_patterns/
                                              # require_approval/path_jail/max_timeout
    def check(self, tool: str, args: dict) -> tuple
        # ("allow"|"ask"|"deny", reason, rule_id)
    @classmethod
    def default(cls) -> "ExecPolicy"

def redact(text: str) -> str      # 脱敏 token/password/私钥/邀请码

# ---- 审计哈希链 ----
class AuditChain:
    def __init__(self, path: str)
    def append(self, event: dict) -> str      # 返回本条 hash，链式 prev
    def verify(self) -> tuple                 # (ok: bool, bad_index: int or -1)
    def tail(self, n: int = 20) -> list

# ---- 指标 ----
class Metrics:
    def inc(self, name: str, value: int = 1, **labels)
    def set_gauge(self, name: str, value, **labels)
    def observe(self, name: str, value: float, **labels)
    def snapshot(self) -> dict
    def render_prometheus(self) -> str

# ---- TLS 自签（可选，调用 openssl；不可用时返回 None）----
def ensure_self_signed_cert(out_dir: str, common_name: str = None) -> tuple
    # (cert_path, key_path) 或 None
def cert_fingerprint(cert_path: str) -> str   # "sha256:AB:CD:..."

# ---- 持久化 / 杂项 ----
def atomic_write(path: str, data, mode: int = 0o600) -> None
def append_jsonl(path: str, obj: dict) -> None       # 带 fsync
def read_jsonl(path: str, limit: int = 0) -> list
def now_ms() -> int
def constant_time_eq(a: str, b: str) -> bool
def check_clock_skew(peer_ts: float, max_skew_s: int = 300) -> tuple  # (ok, skew_s)
```

## 2. 跨平台安装器 `scripts/fleet_install.py`（归属：platform-install）

单文件、纯标准库、Python 3.6+。CLI 与库 API：

```python
VERSION = "5.2.4"

def detect_platform() -> dict
    # {"os","arch","distro","init","is_root","is_termux","is_container",
    #  "python","has_systemd","service_manager","warnings":[...]}

def plan(opts: dict) -> dict
    # {"actions":[{"id","kind","description","requires_root"}],
    #  "warnings":[...], "platform":{...}}

def install(opts: dict) -> dict
    # {"ok":bool,"platform":{},"service":"systemd|openrc|procd|launchd|
    #  "schtasks|termux|docker|nohup","paths":{},"node":{},
    #  "join_uri":"...","next_steps":[...],"errors":[...]}
def uninstall(opts: dict) -> dict
def status(opts: dict) -> dict
def main(argv=None) -> int
```

支持平台适配器（内部类，可单测）：`systemd / openrc / procd(OpenWrt) / launchd /
schtasks(Windows) / termux(sv) / docker / nohup`；Synology DSM / fnOS / QNAP / UGREEN
等无 systemd 的 NAS 走 `nohup + 定时任务提示` 或 `synology` 任务计划文件。

CLI：
```
python3 fleet_install.py --name nas --port 3100 [--dir DIR] [--json] [--dry-run]
  [--join fleet://join/...] [--relay] [--tls auto|off] [--read-only]
  [--user fleet|root] [--no-service] [--uninstall] [--status] [--print-unit]
  [--data-dir DIR] [--source-dir DIR]
```
- `--json` 输出**单行** JSON（便于脚本/agent 解析）；非 json 输出中文人读摘要。
- `--dry-run` 不落盘、不启服务，只打印计划。
- 安装步骤：检测 → 依赖检查（缺失给安装命令，不擅自装）→ 生成数据目录与配置 →
  生成身份（若 `fleet_core` 可用则调用，否则留空）→ 生成服务单元 → 启动 →
  健康检查（HTTP `/health`，3 次退避）→ 打印邀请 URI / 下一步。
- 服务单元安全加固：systemd `NoNewPrivileges=yes`、`PrivateTmp=yes`、
  `ProtectSystem=strict`（数据目录 ReadWritePaths）、`Restart=always`、
  `RestartSec=3`、`WatchdogSec` 可选；Windows 计划任务 + 可选 NSSM；
  launchd `KeepAlive`；procd `respawn`；Termux `sv` + `termux-wake-lock`。
- 卸载安全：只删自己生成的路径（写 manifest 到数据目录），**不删除**数据目录，
  除非 `--purge`；拒绝删除 `/`、`$HOME`、`/usr` 等危险路径。

`scripts/install.sh`：重写为 **POSIX sh 薄引导**（≤ 250 行，不内嵌 Python 源码）：
- 支持 `curl|wget`；`--offline <dir>` 用本地源码；`--version` 固定版本；下载后
  **SHA256 校验**（`SHA256SUMS`）；校验失败拒绝执行；`--dry-run` 透传；
- 自动识别 Termux/OpenWrt/Alpine 并给出对应 python 安装命令；
- 最终 `exec python3 "$SRC/scripts/fleet_install.py" "$@"`。
- 保留旧参数兼容（`--name/--port/--mode/--generate-token/--require-auth/--uninstall` 等）。

`scripts/service-templates/`：`fleet.service`、`fleet.openrc`、`fleet.procd`、
`com.mcpfleet.node.plist`、`fleet-compose.yml`、`Dockerfile`。安装器读模板做变量替换。

`scripts/make-offline.py`：改为打包 `dist/mcp-fleet-offline-<ver>.tar.gz`（源码 +
依赖检查脚本 + SHA256SUMS），不再把 systemd 单元写死。

测试 `tests/test_installer_51.py`：平台检测（monkeypatch）、各适配器生成的单元内容含
关键加固项、dry-run JSON 可解析、路径安全（拒绝 `/`）、join URI 解析、幂等安装、
卸载不越界。≥ 25 用例。

文档 `references/platform-matrix.md`：Android/Termux、飞牛 fnOS、群晖 DSM、威联通、
绿联、OpenWrt/软路由、Debian/Ubuntu、Alpine、macOS、Windows、Docker、云主机
（阿里/腾讯/华为 ECS）的**一条命令**、前置依赖、开机自启方式、已知限制、验证方法。

## 3. Agent 大脑 `scripts/fleet_agent.py`（归属：agent-brain）

单文件、纯标准库。目标：给任意 MCP agent 内置「操作手册 + 自对话协议 + 结构化委派」。

```python
VERSION = "5.2.4"

def build_instructions(cfg: dict = None) -> str
    # MCP initialize 的 instructions：Markdown，≤ 6000 字符，含
    # 何时用本节点、工具选择表、安全边界、自对话四步法、失败升级路径

PROMPTS: dict   # name -> {"description","arguments":[{"name","required"}],"template":str}
# 必须包含：fleet-playbook / fleet-diagnose / fleet-plan / fleet-self-review /
#           fleet-delegate / fleet-incident / fleet-onboard
def list_prompts() -> list
def get_prompt(name: str, arguments: dict = None) -> dict
    # {"description","messages":[{"role":"user","content":{"type":"text","text":...}}]}

SELF_DIALOGUE_TOOLS: list   # [{"name","description","inputSchema"}, ...]
SELF_DIALOGUE_FUNCS: dict   # name -> fn(args, ctx) -> {"content":[{"type":"text","text"}],
                            #                     "isError": False, "structuredContent": {...}}
# 工具：fleet_help / fleet_think / fleet_plan / fleet_reflect / fleet_review /
#       fleet_handoff
# 每个工具的参数见文档；plan 步骤校验（每步需 goal/tool/assignee/acceptance）；
# think/reflect 追加到任务日志（JSONL，0600），返回结构化下一步提示。

class TaskStore:
    def __init__(self, base_dir: str)
    def create_task(self, task: dict) -> dict
    def get_task(self, task_id: str) -> dict
    def append_event(self, task_id: str, event: dict) -> None
    def list_tasks(self, status: str = None) -> list

# 结构化 agent 间消息协议（与邮箱消息 text 字段配合）
def make_envelope(kind: str, thread_id: str, from_, to, payload: dict,
                  task_id: str = None, deadline_s: int = None) -> dict
    # kind ∈ propose|accept|counter|progress|result|fail|review|handoff|hello
def parse_envelope(text: str) -> dict   # 兼容普通文本（返回 {"kind":"text",...}）
def validate_envelope(env: dict) -> tuple  # (ok, error)

class FleetClient:  # 协议无关的薄封装，集成方注入 call_tool(name,args) 可调用对象
    def __init__(self, call_tool)
    def caps(self); def peers(self); def exec_on(self, node, cmd, timeout=60)
    def send_msg(self, to, text, topic=None); def recv_msg(self, me, wait=0)
    def call(self, node, tool, args)   # 经 peer_call

class Orchestrator:
    # 确定性的「规划-委派-验证-汇报」循环，不依赖 LLM；
    # 需要自然语言理解时调用可选的 LLMAdapter（OpenAI 兼容，可 mock）
    def __init__(self, store: TaskStore, client: FleetClient,
                 llm=None, caps_index=None, max_steps=8, max_seconds=120)
    def run(self, task: dict, dry_run: bool = False) -> dict
        # 返回 {"task_id","status","steps":[{"step","tool","node","ok","detail"}],
        #       "result","reflections":[...],"delegated":[...]}
    def resume(self, task_id: str) -> dict

class LLMAdapter:
    # 可选：POST {base_url}/chat/completions，纯 urllib，超时/重试/预算上限
    def __init__(self, base_url, model, api_key="", timeout=30, max_calls=8)
    def complete(self, messages) -> str
```
